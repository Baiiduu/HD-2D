cmake_minimum_required(VERSION 3.22)
project(hd2d_swarm LANGUAGES CXX)
if(NOT GODOT_CPP_SOURCE OR NOT GODOTCPP_CUSTOM_API_FILE)
    message(FATAL_ERROR "Use build.ps1 with the engine API and pinned godot-cpp checkout")
endif()
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
add_subdirectory("${GODOT_CPP_SOURCE}" godot-cpp)
add_library(hd2d_swarm SHARED swarm_kernel.cpp)
target_link_libraries(hd2d_swarm PRIVATE godot-cpp)
target_compile_features(hd2d_swarm PRIVATE cxx_std_17)
set_target_properties(hd2d_swarm PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/bin>"
    LIBRARY_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/bin>")
if(MSVC)
    target_compile_options(hd2d_swarm PRIVATE /O2 /fp:precise /W4)
else()
    target_compile_options(hd2d_swarm PRIVATE -O2 -fno-fast-math -Wall -Wextra)
endif()
